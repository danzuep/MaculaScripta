{{- /* Publishes -data//.html: each chapter as a static HTML fragment (same markup as the reader page, with prefixed ids and a heading) for reader.js to insert while scrolling. */ -}} {{- $testament := . -}} {{- $catalog := index hugo.Data $testament "catalog" -}} {{- $data := partialCached "testament-data.html" $testament $testament -}} {{- range $key, $rows := $data.chapters -}} {{- $parts := split $key "/" -}} {{- $code := index $parts 0 -}} {{- $number := int (index $parts 1) -}} {{- $title := printf "%s %d" (index (where $catalog "code" $code) 0).title $number -}} {{- $html := partial "reader-chapter.html" (dict "testament" $testament "code" $code "number" $number "prefix" (printf "%s.%d." $code $number) "title" $title) -}} {{- $file := resources.FromString (printf "%s-data/%s.html" $testament $key) $html -}} {{- $_ := $file.RelPermalink -}} {{- end -}}
